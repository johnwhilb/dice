flowchart TD

START([战斗开始])

START --> INIT[初始化战斗数据<br/>玩家属性 敌人属性 牌堆]

INIT --> TURN_START[玩家回合开始]


subgraph PLAYER_START["一 回合开始"]

TURN_START --> DICE[5颗骰子重新随机投掷<br/>全部进入可用状态]

DICE --> ENERGY[获得3点能量]

ENERGY --> DRAW[抽取5张牌]

DRAW --> START_EFFECT[结算回合开始效果]

START_EFFECT --> CHECK_START{战斗结束?}

end


CHECK_START -->|否| ACTION
CHECK_START -->|是| RESULT


subgraph PLAYER_ACTION["二 玩家行动阶段"]

ACTION[选择行动]


ACTION --> CARD[打出卡牌]

ACTION --> POWER[使用能量]

ACTION --> END_TURN[结束玩家回合]


CARD --> CHECK_COST{骰子费用满足?}

CHECK_COST -->|否| ACTION

CHECK_COST -->|是| PAY[支付骰子费用]

PAY --> DISABLE[对应骰子进入失活状态]

DISABLE --> RESOLVE[卡牌进入结算区]

RESOLVE --> EFFECT[执行卡牌效果]

EFFECT --> CARD_RESULT[处理卡牌结果<br/>弃牌 消耗 移除]

CARD_RESULT --> CHECK_CARD_END{战斗结束?}

CHECK_CARD_END -->|否| ACTION

CHECK_CARD_END -->|是| RESULT



POWER --> ENERGY_CHOICE{选择能量操作}


ENERGY_CHOICE --> DRAW_CARD[消耗1点能量<br/>抽1张牌]

DRAW_CARD --> HAS_CARD{有牌可抽?}

HAS_CARD -->|是| HAND_CHECK{手牌少于5张?}

HAND_CHECK -->|是| ADD_HAND[加入手牌]

HAND_CHECK -->|否| DISCARD_CARD[进入弃牌堆]

ADD_HAND --> ACTION

DISCARD_CARD --> ACTION

HAS_CARD -->|否| ACTION



ENERGY_CHOICE --> REROLL[消耗2点能量<br/>指定1颗可用骰子重投]

REROLL --> ACTION


ENERGY_CHOICE --> ACTIVATE[消耗2点能量<br/>指定1颗失活骰子<br/>随机激活并重投]

ACTIVATE --> ACTION


end


subgraph PLAYER_END["三 玩家回合结束"]

END_TURN --> DISCARD_HAND[未保留手牌进入弃牌堆]

DISCARD_HAND --> END_EFFECT[结算回合结束效果]

END_EFFECT --> CLEAR_ENERGY[清空剩余能量]

CLEAR_ENERGY --> RESET_DICE[恢复失活骰子可用状态]

RESET_DICE --> CHECK_END{战斗结束?}

end


CHECK_END -->|是| RESULT

CHECK_END -->|否| ENEMY


subgraph ENEMY_TURN["四 敌方回合"]

ENEMY[敌方回合开始]

ENEMY --> ENEMY_START[结算敌方开始效果]

ENEMY_START --> ENEMY_CHECK{战斗结束?}

ENEMY_CHECK -->|否| ATTACK[敌人执行行动]

ATTACK --> ATTACK_CHECK{战斗结束?}

ATTACK_CHECK -->|否| ENEMY_END[结算敌方结束效果]

ENEMY_END --> NEXT[回合数增加1]

NEXT --> TURN_START

end


ENEMY_CHECK -->|是| RESULT

ATTACK_CHECK -->|是| RESULT


RESULT{战斗结果}

RESULT -->|敌人死亡| WIN([战斗胜利])

RESULT -->|玩家死亡| LOSE([战斗失败])
