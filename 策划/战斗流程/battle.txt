
flowchart TD
    A([战斗开始]) --> B[初始化双方属性和牌堆]
    B --> C[玩家回合开始]

    subgraph PLAYER_START["一、玩家回合开始"]
        C --> D[恢复能量至3点]
        D --> E[结算玩家回合开始效果]
        E --> F{战斗是否结束?}
        F -->|否| G[手牌补至5张]
        G --> H[所有骰子随机投掷一次]
        H --> I[获得2次重投机会]
    end

    I --> J

    subgraph PLAYER_ACTION["二、玩家自由行动"]
        J{玩家选择操作}

        J -->|重投骰子| K{剩余重投次数 > 0?}
        K -->|是| L[选择骰子上锁]
        L --> M[重新投掷未上锁的骰子]
        M --> N[重投次数减1]
        N --> J
        K -->|否| J

        J -->|打出手牌| O{骰子条件是否满足?}
        O -->|否| J
        O -->|是| P[将卡牌移入结算区]
        P --> Q[按照JSON顺序结算卡牌效果]
        Q --> R{战斗是否结束?}

        R -->|否| S{能量大于0且抽牌堆有牌?}
        S -->|是| T[消耗1点能量]
        T --> U[自动补充1张手牌]
        U --> V[卡牌进入弃牌堆或消耗区]

        S -->|否| V
        V --> J

        J -->|结束回合| W[结束玩家行动]
    end

    subgraph PLAYER_END["三、玩家回合结束"]
        W --> X[弃掉非保留手牌]
        X --> Y[结算玩家回合结束效果]
        Y --> Z{战斗是否结束?}
    end

    Z -->|否| AA

    subgraph ENEMY_TURN["四、敌人回合"]
        AA[敌人回合开始]
        AA --> AB[结算敌人开局效果]
        AB --> AC{战斗是否结束?}
        AC -->|否| AD[敌人发动攻击]
        AD --> AE{战斗是否结束?}
        AE -->|否| AF[结算敌人回合结束效果]
        AF --> AG{战斗是否结束?}
    end

    AG -->|否| AH[回合数加1]
    AH --> C

    F -->|是| RESULT
    R -->|是| RESULT
    Z -->|是| RESULT
    AC -->|是| RESULT
    AE -->|是| RESULT
    AG -->|是| RESULT

    RESULT{战斗结果}
    RESULT -->|敌人死亡| WIN([战斗胜利])
    RESULT -->|玩家死亡| LOSE([战斗失败])
